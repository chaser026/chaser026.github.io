# 5.7 端到端训练流程实现

> [!NOTE]
> **本节目标**
> - 理解 Transformer 语言模型从数据加载到参数更新的完整训练循环
> - 掌握 Cosine 学习率调度、梯度裁剪、检查点恢复等关键工程实践
> - 能够通过命令行参数配置消融实验（Ablation）并使用 WandB 监控训练过程

---

## 训练流程概览

一次完整的训练迭代包含以下八个步骤，按顺序执行：

1. **同步进度**：根据当前步数 $it$ 计算对应的学习率（Cosine Schedule）
2. **准备数据**：从内存映射文件（memmap）中随机采样一个批次（Batch）
3. **前向传播**：计算 Logits，并通过交叉熵函数算出损失（Loss）
4. **清理现场**：清空优化器中上一轮残余的梯度（`zero_grad`）
5. **反向传播**：计算当前参数的梯度（`loss.backward()`）
6. **安全控制**：执行梯度裁剪（Gradient Clipping），防止权重更新过猛
7. **参数更新**：优化器根据梯度修改模型权重（`optimizer.step()`）
8. **状态监控**：定期在验证集上跑分，并将结果记录到云端（WandB）

> [!TIP]
> 上述流程是大语言模型训练的标准范式，几乎所有基于 PyTorch 的自回归模型训练都遵循此模板。

## 完整实现代码

```python
import argparse
import os
import torch
import numpy as np
import wandb  # 建议安装：pip install wandb
from cs336_basics.nn import TransformerLM
from cs336_basics.optimizer import AdamW, clip_gradient_norm
from cs336_basics.scheduler import get_lr_cosine_schedule
from cs336_basics.data import get_batch
from cs336_basics.checkpointing import save_checkpoint, load_checkpoint
from cs336_basics.losses import cross_entropy

def main():
    parser = argparse.ArgumentParser()
    # --- 模型基础超参数 ---
    parser.add_argument("--batch_size", type=int, default=32)
    parser.add_argument("--context_length", type=int, default=256)
    parser.add_argument("--d_model", type=int, default=512)
    parser.add_argument("--num_layers", type=int, default=4)
    parser.add_argument("--num_heads", type=int, default=8)
    parser.add_argument("--d_ff", type=int, default=2048)
    parser.add_argument("--vocab_size", type=int, default=10000)
    
    # --- 实验/消融 (Ablation) 开关 ---
    # Ablation 1: 移除 RMSNorm
    parser.add_argument("--no_rms_norm", action="store_true", help="Disable RMSNorm completely")
    # Ablation 2: Pre-norm vs Post-norm
    parser.add_argument("--norm_mode", type=str, default="pre", choices=["pre", "post"], help="Normalization placement")
    # Ablation 3: 移除 RoPE (NoPE)
    parser.add_argument("--no_rope", action="store_true", help="Disable Rotary Positional Embeddings")
    # Ablation 4: SwiGLU vs SiLU
    parser.add_argument("--ffn_type", type=str, default="swiglu", choices=["swiglu", "silu"], help="Type of Feed-Forward Network")

    # --- 优化器超参数 ---
    parser.add_argument("--lr", type=float, default=6e-4)
    parser.add_argument("--max_iters", type=int, default=10000)
    parser.add_argument("--warmup_iters", type=int, default=1000)
    parser.add_argument("--min_lr", type=float, default=6e-5)
    parser.add_argument("--max_norm", type=float, default=1.0)
    
    # --- 路径与系统 ---
    parser.add_argument("--train_data_path", type=str, required=True)
    parser.add_argument("--valid_data_path", type=str, required=True)
    parser.add_argument("--out_dir", type=str, default="out")
    parser.add_argument("--device", type=str, default="cuda" if torch.cuda.is_available() else "cpu")

    # --- WandB 设置 ---
    parser.add_argument("--run_name", type=str, default=None, help="WandB 实验名称")
    
    args = parser.parse_args()

    os.makedirs(args.out_dir, exist_ok=True)

    # 1. 加载数据 (使用 memmap)
    # 假设数据是以 uint16 存储的二进制文件
    if not os.path.exists(args.train_data_path):
        raise FileNotFoundError(f"Training data not found at {args.train_data_path}")
    if not os.path.exists(args.valid_data_path):
        raise FileNotFoundError(f"Validation data not found at {args.valid_data_path}")

    # np.memmap 延迟加载数据到内存，非常适合大数据集，并且将二进制文件转为 dtype （uint16） 数组
    train_data = np.memmap(args.train_data_path, dtype=np.uint16, mode='r')
    val_data = np.memmap(args.valid_data_path, dtype=np.uint16, mode='r')

    print(f"训练集大小: {len(train_data)} tokens")
    print(f"验证集大小: {len(val_data)} tokens")

    # 2. 处理消融实验逻辑
    # 如果 no_rope 为 True，则 theta 设为 None，TransformerBlock 内部就不会初始化 RoPE
    actual_rope_theta = None if args.no_rope else 10000.0
    # use_rms_norm 逻辑取反
    use_rms_norm = not args.no_rms_norm

    # 3. 初始化模型
    model = TransformerLM(
        vocab_size=args.vocab_size, 
        context_length=args.context_length,
        d_model=args.d_model, 
        num_layers=args.num_layers,
        num_heads=args.num_heads, 
        d_ff=args.d_ff,
        rope_theta=actual_rope_theta,
        device=args.device,
        # 传入实验参数
        use_rms_norm=use_rms_norm,
        norm_mode=args.norm_mode,
        ffn_type=args.ffn_type
    ).to(args.device)

    print(f"Model Config: Norm={args.norm_mode}, UseNorm={use_rms_norm}, FFN={args.ffn_type}, RoPE={not args.no_rope}")

    # 4. 初始化优化器
    optimizer = AdamW(model.parameters(), lr=args.lr, weight_decay=0.1)

    # 5. 检查点恢复逻辑
    start_iter = 0
    ckpt_path = os.path.join(args.out_dir, "ckpt.pt")
    if os.path.exists(ckpt_path):
        start_iter = load_checkpoint(ckpt_path, model, optimizer)
        print(f"Resuming from iteration {start_iter}")

    # 6. 初始化 WandB 监控
    wandb.init(
        project="cs336-assignment1",
        name=args.run_name, 
        config=args
    )

    # 7. 主训练循环
    for it in range(start_iter, args.max_iters):
        # A. 更新学习率
        lr = get_lr_cosine_schedule(it, args.lr, args.min_lr, args.warmup_iters, args.max_iters)
        for param_group in optimizer.param_groups:
            param_group['lr'] = lr

        # B. 训练步
        model.train()
        x, y = get_batch(train_data, args.batch_size, args.context_length, args.device)
        
        logits = model(x)
        loss = cross_entropy(logits, y)
        
        optimizer.zero_grad()
        loss.backward()
        
        # 梯度裁剪
        clip_gradient_norm(model.parameters(), args.max_norm)
        
        optimizer.step()

        # C. 验证与日志记录
        if it % 100 == 0 or it == args.max_iters - 1:
            model.eval()
            with torch.no_grad():
                vx, vy = get_batch(val_data, args.batch_size, args.context_length, args.device)
                v_logits = model(vx)
                v_loss = cross_entropy(v_logits, vy)
                print(f"Iter {it}: train_loss {loss.item():.4f}, val_loss {v_loss.item():.4f}, lr {lr:.2e}")
                wandb.log({
                    "train/loss": loss.item(), 
                    "val/loss": v_loss.item(), 
                    "lr": lr, 
                    "iter": it + 1
                })

        # D. 保存检查点 (每 1000 步保存一次)
        if it % 1000 == 0 and it > 0:
            save_checkpoint(model, optimizer, it, ckpt_path)

    # 训练结束保存最终模型
    save_checkpoint(model, optimizer, args.max_iters, os.path.join(args.out_dir, "ckpt_final.pt"))
    wandb.finish()

if __name__ == "__main__":
    main()
```

## 关键组件详解

### 数据加载：内存映射文件

训练数据以 `uint16` 格式的二进制文件存储，使用 `np.memmap` 进行延迟加载：

- **优势**：不会一次性将整个数据集读入内存，适合超大语料
- **模式**：`mode='r'` 表示只读
- **数据类型**：`dtype=np.uint16`，每个 token 占 2 字节

> [!WARNING]
> 使用 memmap 时请确保文件路径正确且文件未被其他进程占用，否则会抛出 `FileNotFoundError`。

### 模型初始化与消融实验

模型通过 `TransformerLM` 类实例化，支持四组消融实验开关：

| 参数 | 选项 | 作用 |
| --- | --- | --- |
| `--no_rms_norm` | 布尔开关 | 完全禁用 RMSNorm |
| `--norm_mode` | `pre` / `post` | 控制归一化层放置位置（Pre-norm vs Post-norm） |
| `--no_rope` | 布尔开关 | 禁用旋转位置编码（RoPE） |
| `--ffn_type` | `swiglu` / `silu` | 选择前馈网络激活函数类型 |

关键逻辑：

- `actual_rope_theta = None if args.no_rope else 10000.0`：当禁用 RoPE 时传入 `None`，`TransformerBlock` 内部不会初始化位置编码
- `use_rms_norm = not args.no_rms_norm`：命令行参数与内部变量逻辑取反

### 优化器与学习率调度

- **优化器**：`AdamW`，权重衰减系数 `weight_decay=0.1`
- **学习率调度**：Cosine Schedule，由 `get_lr_cosine_schedule` 函数计算

$$
lr(it) = \text{get\_lr\_cosine\_schedule}(it,\; lr_{\max},\; lr_{\min},\; T_{\text{warmup}},\; T_{\max})
$$

其中：
- $lr_{\max} = 6 \times 10^{-4}$（默认）
- $lr_{\min} = 6 \times 10^{-5}$（默认）
- $T_{\text{warmup}} = 1000$（默认）
- $T_{\max} = 10000$（默认）

> [!NOTE]
> 每个训练步都会手动将计算出的学习率写入 `optimizer.param_groups`，而非使用 PyTorch 内置的 scheduler 对象。

### 梯度裁剪

在 `loss.backward()` 之后、`optimizer.step()` 之前执行：

```python
clip_gradient_norm(model.parameters(), args.max_norm)
```

- **默认阈值**：`max_norm = 1.0`
- **目的**：防止梯度范数过大导致训练不稳定（尤其在训练初期或使用大学习率时）

### 检查点管理

| 时机 | 文件路径 | 说明 |
| --- | --- | --- |
| 训练恢复时 | `out/ckpt.pt` | 若文件存在则从对应迭代恢复 |
| 每 1000 步 | `out/ckpt.pt` | 覆盖式保存，保留最新检查点 |
| 训练结束 | `out/ckpt_final.pt` | 保存最终模型状态 |

### 验证与日志

- **频率**：每 100 步或最后一步执行验证
- **模式**：`model.eval()` + `torch.no_grad()` 确保不计算梯度、不更新 BatchNorm 统计量
- **记录指标**：`train/loss`、`val/loss`、`lr`、`iter` 上报至 WandB
